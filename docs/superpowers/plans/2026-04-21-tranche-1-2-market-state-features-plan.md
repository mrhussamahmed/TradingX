```python
import numpy as np


def forecast_volatility(
    returns: pd.Series,
    *,
    lookback_days: int,
    forecast_method: str = "realized",
    ewma_lambda: float = 0.94,
    garch_train_size: int = 252,
    har_rv_components: tuple[int, int, int] = (1, 5, 22),
) -> pd.Series:
    realized = realized_vol(returns, lookback_days=lookback_days)
    if forecast_method == "realized":
        return realized
    if forecast_method == "ewma":
        return _ewma_vol(returns, ewma_lambda=ewma_lambda)
    if forecast_method == "garch":
        garch = _garch_vol(returns, garch_train_size=garch_train_size)
        return garch.combine_first(realized)
    if forecast_method == "har_rv":
        return _har_rv_vol(
            returns,
            lookback_days=lookback_days,
            components=har_rv_components,
        )
    raise ValueError(f"Unsupported forecast_method: {forecast_method}")


def _har_rv_vol(
    returns: pd.Series,
    *,
    lookback_days: int,
    components: tuple[int, int, int],
) -> pd.Series:
    """HAR-RV forecast per Corsi (2009).

    Models realized variance as a linear combination of three
    temporal components (daily, weekly, monthly by default). Fits
    expanding-window OLS at each step to avoid look-ahead bias.
    Returns annualized volatility aligned to the input index.
    """
    rv = realized_vol(returns, lookback_days=lookback_days).astype(float)
    rv_squared = rv.pow(2.0)  # realized variance series (annualized)

    # Component features: lagged variance at horizons d, w, m
    daily_lag, weekly_lag, monthly_lag = components
    df = pd.DataFrame(
        {
            "rv_sq": rv_squared,
            "rv_sq_d": rv_squared.shift(1).rolling(daily_lag).mean(),
            "rv_sq_w": rv_squared.shift(1).rolling(weekly_lag).mean(),
            "rv_sq_m": rv_squared.shift(1).rolling(monthly_lag).mean(),
        }
    ).dropna()

    forecasts = pd.Series(index=returns.index, dtype=float)
    # Expanding-window refit: for each t, fit OLS on all data up to
    # t-1, predict at t. Minimum training size = 2 * monthly_lag so
    # we have a stable fit.
    min_train = max(2 * monthly_lag, 40)
    for position in range(min_train, len(df)):
        train = df.iloc[:position]
        x = train[["rv_sq_d", "rv_sq_w", "rv_sq_m"]].to_numpy()
        y = train["rv_sq"].to_numpy()
        # OLS closed form with intercept
        x_with_intercept = np.column_stack([np.ones(len(x)), x])
        try:
            coefficients, *_ = np.linalg.lstsq(x_with_intercept, y, rcond=None)
        except np.linalg.LinAlgError:
            continue
        next_row = df.iloc[position][["rv_sq_d", "rv_sq_w", "rv_sq_m"]].to_numpy()
        predicted_variance = float(
            coefficients[0] + np.dot(coefficients[1:], next_row)
        )
        if predicted_variance <= 0:
            continue
        target_timestamp = df.index[position]
        forecasts.loc[target_timestamp] = math.sqrt(predicted_variance)
    return forecasts
```

```python
def test_har_rv_recovers_constant_vol_on_stationary_series():
    # Stationary returns with ~constant vol. HAR-RV should forecast
    # close to the true annualized vol once training is sufficient.
    target_daily_vol = 0.012
    expected_annual_vol = target_daily_vol * math.sqrt(252)
    rng = np.random.default_rng(seed=99)
    returns = pd.Series(
        rng.normal(scale=target_daily_vol, size=800),
        index=pd.date_range("2020-01-01", periods=800, freq="B"),
    )
    forecast = forecast_volatility(
        returns,
        lookback_days=20,
        forecast_method="har_rv",
    )
    final_half = forecast.dropna().iloc[-200:]
    assert len(final_half) > 0
    mean_forecast = float(final_half.mean())
    # Tolerate 25% error on stationary series (HAR-RV's OLS fit is
    # noisy on short windows even with constant underlying vol)
    assert abs(mean_forecast - expected_annual_vol) / expected_annual_vol < 0.25
```

```python
def _normalize_treasury_frame(
    frame_by_code: dict[str, pd.DataFrame],
) -> pd.DataFrame:
    """Combine DGS2 and DGS10 frames into a single cache layout."""
    if set(frame_by_code.keys()) != set(TREASURY_SERIES_CODES):
        missing = set(TREASURY_SERIES_CODES) - set(frame_by_code.keys())
        raise ValueError(
            f"Missing Treasury series in fetch response: {sorted(missing)}"
        )
    aligned = pd.concat(
        [frame.rename(columns={code: code.lower()}) for code, frame in frame_by_code.items()],
        axis=1,
    ).dropna()
    aligned.index = pd.to_datetime(aligned.index, utc=True).normalize()
    aligned = aligned.sort_index(kind="stable")
    aligned["spread_2s10s_bps"] = (
        (aligned["dgs10"] - aligned["dgs2"]) * 100.0
    )
    return pd.DataFrame(
        {
            "date": aligned.index.strftime("%Y-%m-%d"),
            "dgs2": aligned["dgs2"].astype(float).values,
            "dgs10": aligned["dgs10"].astype(float).values,
            "spread_2s10s_bps": aligned["spread_2s10s_bps"].astype(float).values,
        }
    ).reset_index(drop=True)


def refresh_treasury_yield_cache(
    cache_path: Path,
    *,
    start_date: date,
    end_date: date,
    now_utc: datetime,
    fetcher: RiskFreeFetcher | None = None,
) -> Path:
    if end_date < start_date:
        raise ValueError("end_date must be on or after start_date.")
    active_fetcher = fetcher or _default_fred_fetcher
    frame_by_code = {
        code: active_fetcher(code, RISK_FREE_SOURCE, start_date, end_date)
        for code in TREASURY_SERIES_CODES
    }
    normalized = _normalize_treasury_frame(frame_by_code)
    cache_path.parent.mkdir(parents=True, exist_ok=True)
    normalized.to_csv(cache_path, index=False)
    timestamp = now_utc.astimezone(UTC).timestamp()
    os.utime(cache_path, (timestamp, timestamp))
    return cache_path


def load_treasury_yield_cache(cache_path: Path) -> pd.DataFrame:
    frame = pd.read_csv(cache_path)
    expected_columns = {"date", "dgs2", "dgs10", "spread_2s10s_bps"}
    if set(frame.columns) != expected_columns:
        raise ValueError(
            "Treasury yield cache must contain exactly "
            "date, dgs2, dgs10, and spread_2s10s_bps columns."
        )
    loaded = frame.copy()
    loaded["date"] = pd.to_datetime(loaded["date"], utc=True).dt.normalize()
    return loaded.sort_values("date", kind="stable").reset_index(drop=True)


def yield_curve_spread_for_date(
    cache_path: Path,
    *,
    as_of_date: date,
) -> Decimal:
    frame = load_treasury_yield_cache(cache_path)
    keyed = frame.set_index("date")["spread_2s10s_bps"]
    target = pd.Timestamp(as_of_date).tz_localize(UTC)
    available = keyed.loc[keyed.index <= target]
    if available.empty:
        raise ValueError(
            f"No cached Treasury spread is available on or before "
            f"{as_of_date.isoformat()}."
        )
    return Decimal(str(float(available.iloc[-1])))
```

```python
"""Rule-based regime classifier.

Produces a deterministic RegimeSnapshot from three sub-indicators:
SMA-200 position, VIX level, and Treasury 2s10s yield-curve shape.
Each indicator can be None (missing data); aggregate rules treat
None as "not fired."
"""

from __future__ import annotations

from datetime import date
from decimal import Decimal

from tradingx.types import RegimeSnapshot


VIX_ELEVATED_THRESHOLD_DEFAULT = Decimal("25")
VIX_STRESS_THRESHOLD_DEFAULT = Decimal("35")


def classify_regime(
    *,
    observation_date: date,
    spy_close: Decimal,
    spy_sma_200: Decimal,
    vix_close: Decimal | None,
    yield_curve_spread_bps: Decimal | None,
    vix_elevated_threshold: Decimal = VIX_ELEVATED_THRESHOLD_DEFAULT,
    vix_stress_threshold: Decimal = VIX_STRESS_THRESHOLD_DEFAULT,
) -> RegimeSnapshot:
    above_sma_200: bool | None = spy_close > spy_sma_200
    vix_elevated: bool | None = (
        vix_close >= vix_elevated_threshold if vix_close is not None else None
    )
    vix_stressed: bool | None = (
        vix_close >= vix_stress_threshold if vix_close is not None else None
    )
    yield_curve_inverted: bool | None = (
        yield_curve_spread_bps < Decimal("0")
        if yield_curve_spread_bps is not None
        else None
    )

    # Aggregate rules evaluated in order. None-valued indicators
    # are treated as "not fired."
    if bool(vix_stressed) or (
        bool(yield_curve_inverted) and not bool(above_sma_200)
    ):
        regime = "stressed"
    elif not bool(above_sma_200) or bool(vix_elevated):
        regime = "risk_off"
    elif bool(above_sma_200) and not bool(vix_elevated):
        regime = "risk_on"
    else:
        regime = "neutral"

    return RegimeSnapshot(
        observation_date=observation_date,
        regime=regime,
        above_sma_200=above_sma_200,
        vix_elevated=vix_elevated,
        vix_stressed=vix_stressed,
        yield_curve_inverted=yield_curve_inverted,
        spy_close=spy_close,
        spy_sma_200=spy_sma_200,
        vix_close=vix_close,
        yield_curve_spread_bps=yield_curve_spread_bps,
    )

    "VIX_STRESS_THRESHOLD_DEFAULT",

)
```

```python
def _forecast_error_metrics(
    bars_by_symbol: dict[str, pd.DataFrame],
    forecasts: dict[str, dict[str, pd.Series]],
    *,
    lookback_days: int = 20,
) -> dict[str, dict[str, object]]:
    """Score method forecasts against forward realized volatility.

    Target (at index t): annualized std of returns over the forward
    window [t+1, ..., t+lookback_days]. Computed as a backward
    rolling std shifted left by `lookback_days` so the value at
    position t comes from the window that starts at t+1 and ends at
    t+lookback_days, then scaled by sqrt(252) for annualization.

    This replaces the prior variance-unit target
    (`pct_change().pow(2.0)`); all pooled and per-symbol metrics are
    now in annualized volatility units.
    """
    metrics: dict[str, dict[str, object]] = {}
    for method_name, method_series in forecasts.items():
        symbol_metrics: dict[str, dict[str, float | int]] = {}
        pooled_abs_errors: list[float] = []
        pooled_squared_errors: list[float] = []
        pooled_qlike: list[float] = []
        for symbol, forecast in method_series.items():
            returns = bars_by_symbol[symbol]["close"].astype(float).pct_change()
            forward_realized_vol = (
                returns.rolling(lookback_days).std(ddof=1).shift(-lookback_days)
                * math.sqrt(252.0)
            )
            valid = forecast.notna() & forward_realized_vol.notna()
            valid_index = forecast.loc[valid].index
            if valid_index.empty:
                symbol_metrics[symbol] = {
                    "prediction_count": 0,
                    "mae_vol": float("nan"),
                    "rmse_vol": float("nan"),
                    "msfe_vol": float("nan"),
                    "qlike": float("nan"),
                }
                continue
            aligned_forecast = forecast.loc[valid_index]
            aligned_target = forward_realized_vol.loc[valid_index]
            residuals = aligned_forecast - aligned_target
            abs_errors = residuals.abs()
            squared_errors = residuals.pow(2.0)

            # QLIKE derived from the vol-unit square so the
            # headline loss and QLIKE share the same input space.
            forecast_variance = aligned_forecast.pow(2.0).clip(lower=1e-12)
            actual_variance = aligned_target.pow(2.0)
            qlike = (actual_variance / forecast_variance) + forecast_variance.map(
                math.log
            )

            symbol_metrics[symbol] = {
                "prediction_count": int(len(valid_index)),
                "mae_vol": float(abs_errors.mean()),
                "rmse_vol": float(math.sqrt(squared_errors.mean())),
                "msfe_vol": float(squared_errors.mean()),
                "qlike": float(qlike.mean()),
            }
            pooled_abs_errors.extend(float(v) for v in abs_errors.tolist())
            pooled_squared_errors.extend(float(v) for v in squared_errors.tolist())
            pooled_qlike.extend(float(v) for v in qlike.tolist())

        if pooled_squared_errors:
            pooled_rmse = math.sqrt(
                sum(pooled_squared_errors) / len(pooled_squared_errors)
            )
            pooled_msfe = sum(pooled_squared_errors) / len(pooled_squared_errors)
        else:
            pooled_rmse = float("nan")
            pooled_msfe = float("nan")
        metrics[method_name] = {
            "prediction_count": len(pooled_abs_errors),
            "pooled_mae_vol": (
                sum(pooled_abs_errors) / len(pooled_abs_errors)
                if pooled_abs_errors
                else float("nan")
            ),
            "pooled_rmse_vol": pooled_rmse,
            "pooled_msfe_vol": pooled_msfe,
            "pooled_qlike": (
                sum(pooled_qlike) / len(pooled_qlike)
                if pooled_qlike
                else float("nan")
            ),
            "symbols": symbol_metrics,
        }
    return metrics
```

```python
def _pick_aggregate_winner(
    metrics: dict[str, dict[str, object]],
) -> str | None:
    """Return the method with the lowest pooled_rmse_vol among
    methods that produced at least one valid prediction. Returns
    None if no method is valid.
    """
    best_method: str | None = None
    best_rmse = float("inf")
    for method, method_metrics in metrics.items():
        prediction_count = int(method_metrics.get("prediction_count", 0))
        if prediction_count <= 0:
            continue
        rmse = float(method_metrics.get("pooled_rmse_vol", float("nan")))
        if math.isnan(rmse) or math.isinf(rmse):
            continue
        if rmse < best_rmse:
            best_rmse = rmse
            best_method = method
    return best_method
```
